const history={};
function ping(url){
  return new Promise(r=>{
    const t=performance.now();
    const img=new Image();
    const tm=setTimeout(()=>{img.src='';r(999);},6000);
    img.onload=img.onerror=()=>{clearTimeout(tm);r(Math.round(performance.now()-t));};
    img.src=url+'?_='+Date.now()+Math.random();
  });
}
async function reliablePing(url){
  const a=await ping(url);
  await new Promise(r=>setTimeout(r,140));
  const b=await ping(url);
  return Math.round((a+b)/2);
}

function classify(ms){

    tier:'bad',cardCls:'bad',
    phrase:ms>=999?'انقطع الاتصال — ما وصل أي جواب!':'بي لغوة — الوطنية طافية!',
    icon:ms>=999?'⛔':'🔴',
    numCls:'rn-bad',badgeCls:'rb-bad',
    badgeTxt:ms>=999?'⛔ منقطع':'🔴 بطيء',
    phraseCls:'rp-bad',color:'#ff375f'

    tier:'ok',cardCls:'ok',
    phrase:'يگضي شغل — مو طيّارة بس ماشي',

}

function drawSpark(id,color){
  const el=document.getElementById('rcSpark');
  if(!el||!history[id])return;
  const vals=history[id];
  const max=Math.max(...vals.filter(v=>v<999),200);
  el.innerHTML='';
  vals.forEach(v=>{
    const pct=v>=999?100:Math.max(8,Math.round((v/max)*100));
    const s=document.createElement('span');
    s.style.cssText=`height:${pct}%;background:${color}80;border-radius:3px;flex:1;`;
    el.appendChild(s);
  });
}
